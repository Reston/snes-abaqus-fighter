sha=lambda raw:hashlib.sha256(raw).hexdigest()
def files(base,directories):
    return {p.relative_to(base).as_posix():sha(p.read_bytes())
            for d in directories for p in sorted((base/d).rglob('*'))
            if p.is_file() and '__pycache__' not in p.parts and p.suffix not in ('.pyc','.pyo')}
def digest_map(mapping):
    return sha(json.dumps(mapping,sort_keys=True,separators=(',',':')).encode())
def counts():
    return {key:value['executions'] for key,value in arena.CHECKS.items()}
def delta(before):
    return {key:row['executions']-before.get(key,0) for key,row in arena.CHECKS.items()
            if row['executions']!=before.get(key,0)}
def artifact(path):
    return {'path':str(path),'sha256':sha(path.read_bytes()),'bytes':path.stat().st_size}

# Read-only Python return observer retains the exact local 'data' already
# hashed by the unchanged Rig.game. No Rig method, controller or state changes.
observed_games={}
game_code=arena.Rig.game.__code__
def observe(frame,event,result):
    if event=='return' and frame.f_code is game_code:
        data=frame.f_locals['data']
        if sha(data)!=result:
            raise AssertionError('Unchanged Rig.game return does not hash its exact data')
        previous=observed_games.setdefault(result,data)
        if previous!=data:
            raise AssertionError('Gameplay digest collision')

def first_difference(expected_digest,r,stage,index,keys):
    expected=observed_games[expected_digest]
    actual=observed_games[r.game()]
    differences=[{'offset':i,'candidate':a,'reference':b}
                 for i,(a,b) in enumerate(zip(expected,actual)) if a!=b]
    words=[{'word':i,'candidate':a,'reference':b}
           for i,(a,b) in enumerate(zip(struct.unpack('<114H',expected[:228]),
                                      struct.unpack('<114H',actual[:228]))) if a!=b]
    return {'stage':stage,'callback':index,'keys':keys,
            'candidate_game_sha256':expected_digest,'reference_game_sha256':sha(actual),
            'candidate_game_bytes_hex':expected.hex(),'reference_game_bytes_hex':actual.hex(),
            'differing_bytes':differences,'differing_trace_words':words,
            'reference_state':r.state()}

def replay_current_source(case,report,reference_symbols):
    require=arena.require
    r=arena.Rig('snes9x',case['stage'],case['fighter_mirror'],reference_symbols,report,
                rom=REFERENCE/'dist/abaqus-after-hours.sfc',monitor=True,captures=CAPTURES)
    start=counts()
    try:
        r.boot()
        same=r.initial_game==case['replay']['initial_game']
        if not same:
            report['first_divergence']=first_difference(case['replay']['initial_game'],r,case['stage'],'initial',[])
        require(same,'Current-source reference starts at identical complete unmasked gameplay state',report.get('first_divergence'))
        initial_frame=r.state()['frame']
        def zero_upload_guard():
            state=r.life()
            require(state[0]==case['stage']+1 and state[1:6]==bytes([0,0,0,6,0])
                    and state[12]==state[14]==0,
                    'Arena-disabled reference clock/phase/queue/arm/serial remain exact initialized values',list(state[:21]))
            require(r.vram()[0x6800:0x6e00]==r.sources[case['stage']][:1536],
                    'Arena-disabled reference entire pool remains exact phase zero')
            require(r.health['ambient_uploads']==0,'Arena-disabled reference completes zero arena DMA')
        zero_upload_guard()
        for index,row in enumerate(case['replay']['inputs']):
            r.tick(row['p1'],row['p2'],record=False)
            actual=r.game()
            same=actual==row['game']
            if not same:
                report['first_divergence']=first_difference(row['game'],r,case['stage'],index,[row['p1'],row['p2']])
            require(same,'Every callback complete unmasked TRACE0..113/Fighter64/Wave16/RNG/private-super equals current-source reference',
                    report.get('first_divergence'))
            zero_upload_guard()
        require(0<r.e.audio_peak<32767 and r.e.audio_energy>0,
                'Reference full-session captured PCM stays active and unclipped',r.e.audio_peak)
        require(r.monitor and r.health['callbacks']>=len(case['replay']['inputs']),
                'Reference retains unchanged strict native health on every replay callback',r.health)
        return {'stage':case['stage'],'fighter_mirror':case['fighter_mirror'],
                'callbacks':len(case['replay']['inputs']),'initial_equal':True,'all_unmasked_equal':True,
                'absolute_TRACE2_included':True,'initial_frame':initial_frame,
                'gameplay_byte_count':len(observed_games[case['replay']['initial_game']]),
                'reference_rom_sha256':REFERENCE_SHA,'arena_dma_count':r.health['ambient_uploads'],
                'pool_phase':0,'strict_native_health':r.health,'checks':delta(start),
                'pcm_peak':r.e.audio_peak,'pcm_energy':r.e.audio_energy}
    finally:
        r.e.close()
